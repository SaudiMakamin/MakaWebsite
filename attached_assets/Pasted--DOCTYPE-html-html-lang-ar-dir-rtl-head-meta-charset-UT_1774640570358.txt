  <script>
    function escapeHtml(str) {
      return String(str || "")
        .replace(/&/g, "&amp;")
        .replace(/</g, "&lt;")
        .replace(/>/g, "&gt;")
        .replace(/"/g, "&quot;")
        .replace(/'/g, "&#039;");
    }

    function normalizeText(value) {
      return String(value || "").trim();
    }

    function showResultSuccess(html) {
      const result = document.getElementById("result");
      result.className = "success";
      result.style.display = "block";
      result.innerHTML = html;
    }

    function showResultError(message) {
      const result = document.getElementById("result");
      result.className = "error";
      result.style.display = "block";
      result.innerHTML = escapeHtml(message || "حدث خطأ غير متوقع.");
    }

    function clearResult() {
      const result = document.getElementById("result");
      result.style.display = "none";
      result.className = "";
      result.innerHTML = "";
    }

    function setLoading(isLoading) {
      const btn = document.getElementById("submitBtn");
      const form = document.getElementById("shareholderForm");

      btn.disabled = isLoading;
      btn.textContent = isLoading ? "جاري إرسال الطلب..." : "إرسال الطلب";

      if (isLoading) {
        form.classList.add("loading");
      } else {
        form.classList.remove("loading");
      }
    }

    function validatePdfFile(file, fieldLabel) {
      if (!file) return true;

      const name = String(file.name || "").toLowerCase();
      const type = String(file.type || "").toLowerCase();

      const isPdf = type === "application/pdf" || name.endsWith(".pdf");
      if (!isPdf) {
        throw new Error(fieldLabel + " يجب أن يكون بصيغة PDF فقط.");
      }

      return true;
    }

    async function fileToBase64(file) {
      if (!file) return null;

      return new Promise((resolve, reject) => {
        const reader = new FileReader();

        reader.onload = function(e) {
          const raw = String(e.target.result || "");
          const parts = raw.split(",");
          const base64 = parts.length > 1 ? parts[1] : "";

          resolve({
            name: file.name,
            mimeType: file.type || "application/pdf",
            data: base64
          });
        };

        reader.onerror = function() {
          reject(new Error("تعذر قراءة الملف: " + (file.name || "")));
        };

        reader.readAsDataURL(file);
      });
    }

    function validateBeforeSubmit() {
      const requiredFields = [
        { id: "shareholderType", label: "نوع المساهم" },
        { id: "fullName", label: "الاسم الكامل / الاسم القانوني للشركة" },
        { id: "idOrCr", label: "رقم الهوية / السجل التجاري" },
        { id: "nationalityOrCountry", label: "الجنسية / دولة التأسيس" },
        { id: "mobile", label: "رقم الجوال" },
        { id: "email", label: "البريد الإلكتروني" },
        { id: "address", label: "العنوان" },
        { id: "founderShareholder", label: "المساهم المؤسس" },
        { id: "sharesNumber", label: "عدد الأسهم" },
        { id: "sharesWords", label: "عدد الأسهم كتابة" },
        { id: "paidAmount", label: "القيمة المدفوعة" },
        { id: "currency", label: "العملة" },
        { id: "bankName", label: "اسم البنك" },
        { id: "accountName", label: "اسم صاحب الحساب" },
        { id: "iban", label: "IBAN" },
        { id: "bankCountry", label: "دولة البنك" }
      ];

      for (const field of requiredFields) {
        const el = document.getElementById(field.id);
        if (!el || !normalizeText(el.value)) {
          throw new Error("الحقل مطلوب: " + field.label);
        }
      }

      const email = normalizeText(document.getElementById("email").value);
      const emailOk = /^[^\s@]+@[^\s@]+\.[^\s@]+$/.test(email);
      if (!emailOk) {
        throw new Error("البريد الإلكتروني غير صالح.");
      }

      const mainDocument = document.getElementById("mainDocument").files[0];
      const bankDocument = document.getElementById("bankDocument").files[0];
      const supportDocument = document.getElementById("supportDocument").files[0];

      if (!mainDocument) throw new Error("المستند الرئيسي مطلوب.");
      if (!bankDocument) throw new Error("المستند البنكي مطلوب.");

      validatePdfFile(mainDocument, "المستند الرئيسي");
      validatePdfFile(bankDocument, "المستند البنكي");
      if (supportDocument) validatePdfFile(supportDocument, "المستند الداعم");

      if (!document.getElementById("declaration").checked) {
        throw new Error("يجب الإقرار بصحة البيانات قبل الإرسال.");
      }
    }

    async function submitForm() {
      clearResult();

      try {
        validateBeforeSubmit();
        setLoading(true);

        const mainDocument = document.getElementById("mainDocument").files[0];
        const bankDocument = document.getElementById("bankDocument").files[0];
        const supportDocument = document.getElementById("supportDocument").files[0];

        const data = {
          shareholderType: normalizeText(document.getElementById("shareholderType").value),
          fullName: normalizeText(document.getElementById("fullName").value),
          idOrCr: normalizeText(document.getElementById("idOrCr").value),
          nationalityOrCountry: normalizeText(document.getElementById("nationalityOrCountry").value),
          dateOfBirth: normalizeText(document.getElementById("dateOfBirth").value),
          authorizedPerson: normalizeText(document.getElementById("authorizedPerson").value),
          authorizedPersonId: normalizeText(document.getElementById("authorizedPersonId").value),
          mobile: normalizeText(document.getElementById("mobile").value),
          email: normalizeText(document.getElementById("email").value),
          address: normalizeText(document.getElementById("address").value),
          founderShareholder: normalizeText(document.getElementById("founderShareholder").value),
          sharesNumber: normalizeText(document.getElementById("sharesNumber").value),
          sharesWords: normalizeText(document.getElementById("sharesWords").value),
          paidAmount: normalizeText(document.getElementById("paidAmount").value),
          currency: normalizeText(document.getElementById("currency").value),
          bankName: normalizeText(document.getElementById("bankName").value),
          accountName: normalizeText(document.getElementById("accountName").value),
          iban: normalizeText(document.getElementById("iban").value),
          bankCountry: normalizeText(document.getElementById("bankCountry").value),
          updateMobile: document.getElementById("updateMobile").checked,
          updateEmail: document.getElementById("updateEmail").checked,
          updateAddress: document.getElementById("updateAddress").checked,
          updateBank: document.getElementById("updateBank").checked,
          updateShareholding: document.getElementById("updateShareholding").checked,
          notes: normalizeText(document.getElementById("notes").value),
          mainDocument: await fileToBase64(mainDocument),
          bankDocument: await fileToBase64(bankDocument),
          supportDocument: supportDocument ? await fileToBase64(supportDocument) : null
        };

        google.script.run
          .withSuccessHandler(function(res) {
            setLoading(false);

            showResultSuccess(
              "<strong>" + escapeHtml(res && res.message ? res.message : "تم إرسال الطلب بنجاح.") + "</strong><br>" +
              "الرقم المرجعي للطلب: <strong>" + escapeHtml(res && res.requestId ? res.requestId : "") + "</strong>"
            );

            document.getElementById("shareholderForm").reset();
            window.scrollTo({ top: document.body.scrollHeight, behavior: "smooth" });
          })
          .withFailureHandler(function(err) {
            setLoading(false);
            showResultError("تعذر إرسال الطلب: " + (err && err.message ? err.message : "خطأ غير معروف."));
          })
          .submitForm(data);

      } catch (e) {
        setLoading(false);
        showResultError(e && e.message ? e.message : "تعذر تجهيز الملفات أو إرسال الطلب.");
      }
    }
